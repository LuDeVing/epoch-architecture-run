class RMSNorm(nn.Module):
    def __init__(self, d: int, eps: float = 1e-6):
        super().__init__()
        self.weight = nn.Parameter(torch.ones(d))
        self.eps = eps

    def forward(self, x):
        return F.rms_norm(x, (x.size(-1),), self.weight, self.eps)


def rope_cache(seq_len: int, head_dim: int, theta: float, device) -> tuple[torch.Tensor, torch.Tensor]:
    inv = 1.0 / (theta ** (torch.arange(0, head_dim, 2, device=device).float() / head_dim))
    t = torch.arange(seq_len, device=device).float()
    freqs = torch.outer(t, inv)                       # (T, D/2)
    return freqs.cos(), freqs.sin()


def apply_rope(x: torch.Tensor, cos: torch.Tensor, sin: torch.Tensor) -> torch.Tensor:
    # x: (B, H, T, D); rotate pairs (x1, x2) = (x[..., :D/2], x[..., D/2:])
    d = x.size(-1) // 2
    x1, x2 = x[..., :d], x[..., d:]
    cos, sin = cos[: x.size(2)].to(x.dtype), sin[: x.size(2)].to(x.dtype)
    return torch.cat([x1 * cos - x2 * sin, x1 * sin + x2 * cos], dim=-1)


class Attention(nn.Module):
    def __init__(self, cfg: Config):
        super().__init__()
        self.n_head = cfg.n_head
        self.hd = cfg.d_model // cfg.n_head
        self.rd = 32
        self.wq = nn.Linear(cfg.d_model, self.n_head * (self.hd + self.rd), bias=False)
        self.compress = nn.Linear(cfg.d_model, 128, bias=False)
        self.latent_norm = RMSNorm(128)
        self.expand = nn.Linear(128, 2 * self.n_head * self.hd, bias=False)
        self.rotary = nn.Linear(cfg.d_model, self.rd, bias=False)
        self.q_norm, self.k_norm = RMSNorm(self.hd), RMSNorm(self.hd)
        self.wo = nn.Linear(self.n_head * self.hd, cfg.d_model, bias=False)
        self.window = cfg.sliding_window

    def forward(self, x, cos, sin):
        B, T, _ = x.shape
        q = self.wq(x).view(B, T, self.n_head, self.hd + self.rd).transpose(1, 2)
        qc, qr = q.split([self.hd, self.rd], dim=-1)
        kv = self.expand(self.latent_norm(self.compress(x)))
        k, v = kv.view(B, T, self.n_head, 2 * self.hd).transpose(1, 2).chunk(2, dim=-1)
        kr = self.rotary(x).view(B, T, 1, self.rd).transpose(1, 2)
        qr, kr = apply_rope(qr, cos, sin), apply_rope(kr, cos, sin)
        q = torch.cat([self.q_norm(qc), qr], dim=-1)
        k = torch.cat([self.k_norm(k), kr.expand(-1, self.n_head, -1, -1)], dim=-1)
        if self.window and self.window < T:
            i = torch.arange(T, device=x.device)
            mask = (i[None, :] <= i[:, None]) & (i[:, None] - i[None, :] < self.window)
            y = F.scaled_dot_product_attention(q, k, v, attn_mask=mask)
        else:
            y = F.scaled_dot_product_attention(q, k, v, is_causal=True)
        return self.wo(y.transpose(1, 2).reshape(B, T, -1))


class MLP(nn.Module):
    """SwiGLU feed-forward."""
    def __init__(self, cfg: Config):
        super().__init__()
        self.w_gate = nn.Linear(cfg.d_model, cfg.d_ff, bias=False)
        self.w_up = nn.Linear(cfg.d_model, cfg.d_ff, bias=False)
        self.w_down = nn.Linear(cfg.d_ff, cfg.d_model, bias=False)

    def forward(self, x):
        return self.w_down(F.silu(self.w_gate(x)) * self.w_up(x))


class Block(nn.Module):
    def __init__(self, cfg: Config):
        super().__init__()
        self.n1, self.n2 = RMSNorm(cfg.d_model), RMSNorm(cfg.d_model)
        self.attn, self.mlp = Attention(cfg), MLP(cfg)
        self.mix_logits = nn.Parameter(torch.eye(4).repeat(2, 1, 1) * 6.0)
        self.read_logits = nn.Parameter(torch.zeros(2, 4))
        self.write_logits = nn.Parameter(torch.tensor([[0.03, -0.01, -0.01, -0.01], [-0.01, 0.03, -0.01, -0.01]]))

    def combine(self, x, branch, stage):
        mixing = self.mix_logits[stage].float().exp()
        for _ in range(20):
            mixing = mixing / mixing.sum(dim=-1, keepdim=True)
            mixing = mixing / mixing.sum(dim=-2, keepdim=True)
        mixed = torch.einsum('ij,btjd->btid', mixing.to(x.dtype), x)
        weights = 2 * self.write_logits[stage].sigmoid()
        return mixed + branch.unsqueeze(2) * weights.to(x.dtype)[None, None, :, None]

    def forward(self, x, cos, sin):
        weights = self.read_logits[0].softmax(dim=-1).to(x.dtype)
        branch = self.attn(self.n1((x * weights[None, None, :, None]).sum(2)), cos, sin)
        x = self.combine(x, branch, 0)
        weights = self.read_logits[1].softmax(dim=-1).to(x.dtype)
        branch = self.mlp(self.n2((x * weights[None, None, :, None]).sum(2)))
        return self.combine(x, branch, 1)


class GPT(nn.Module):
    def __init__(self, cfg: Config):
        super().__init__()
        self.cfg = cfg
        self.tok = nn.Embedding(cfg.vocab_size, cfg.d_model)
        self.blocks = nn.ModuleList([Block(cfg) for _ in range(cfg.n_layer)])
        self.norm = RMSNorm(cfg.d_model)
        self.head = nn.Linear(cfg.d_model, cfg.vocab_size, bias=False)
        if cfg.tie_embeddings:
            self.head.weight = self.tok.weight
        self.apply(self._init)
        for b in self.blocks:                              # scaled residual-branch init
            nn.init.normal_(b.attn.wo.weight, std=0.02 / math.sqrt(2 * cfg.n_layer))
            nn.init.normal_(b.mlp.w_down.weight, std=0.02 / math.sqrt(2 * cfg.n_layer))

    @staticmethod
    def _init(m):
        if isinstance(m, nn.Linear):
            nn.init.normal_(m.weight, std=0.02)
        elif isinstance(m, nn.Embedding):
            nn.init.normal_(m.weight, std=0.02)

    def forward(self, idx, targets=None):
        B, T = idx.shape
        cos, sin = rope_cache(T, 32, self.cfg.rope_theta, idx.device)
        x = self.tok(idx).unsqueeze(2).expand(-1, -1, 4, -1)
        for b in self.blocks:
            x = b(x, cos, sin)
        logits = self.head(self.norm(x.mean(dim=2)))
        if targets is None:
            return logits, None
        # logits stay in the autocast dtype; a float32 copy of (B*T, 50304) alone is 0.8 GB at B=4
        loss = F.cross_entropy(logits.view(-1, logits.size(-1)), targets.view(-1))
        return logits, loss


def param_counts(model: nn.Module) -> tuple[int, int]:
    total = sum(p.numel() for p in model.parameters())
    embed = model.tok.weight.numel()
    return total, total - embed


# ======================================================================================
# Muon optimizer (Newton-Schulz orthogonalized momentum; Jordan et al. 2024)
# ======================================================================================
def newton_schulz5(G: torch.Tensor, steps: int = 5) -> torch.Tensor:
    a, b, c = (3.4445, -4.7750, 2.0315)
    X = G.bfloat16()
    X = X / (X.norm() + 1e-7)
    transposed = G.size(0) > G.size(1)
    if transposed:
        X = X.T
    for _ in range(steps):
        A = X @ X.T
        B_ = b * A + c * A @ A
        X = a * X + B_ @ X
    return (X.T if transposed else X).to(G.dtype)


class Muon(torch.optim.Optimizer):
    def __init__(self, params, lr=0.02, momentum=0.95, weight_decay=0.0, nesterov=True):
        super().__init__(params, dict(lr=lr, momentum=momentum, weight_decay=weight_decay, nesterov=nesterov))

    @torch.no_grad()
    def step(self):
        for group in self.param_groups:
            for p in group["params"]:
                if p.grad is None:
                    continue
                g = p.grad
                st = self.state[p]
                if "buf" not in st:
                    st["buf"] = torch.zeros_like(g)
                buf = st["buf"]
                buf.mul_(group["momentum"]).add_(g)
                g = g.add(buf, alpha=group["momentum"]) if group["nesterov"] else buf
                u = newton_schulz5(g.reshape(g.size(0), -1)).view_as(g)
                scale = max(1.0, g.size(0) / g.size(1)) ** 0.5     # shape-aware scaling
                if group["weight_decay"]:
                    p.mul_(1 - group["lr"] * group["weight_decay"])
                p.add_(u, alpha=-group["lr"] * scale)


def build_optimizers(model: GPT, cfg: Config):
    matrix = [p for n, p in model.named_parameters() if p.ndim == 2 and "tok" not in n and "head" not in n]
    other = [p for n, p in model.named_parameters() if not (p.ndim == 2 and "tok" not in n and "head" not in n)]
    return [Muon(matrix, lr=cfg.lr, weight_decay=cfg.weight_decay),
            torch.optim.AdamW(other, lr=cfg.lr_adam, betas=(0.9, 0.95), weight_decay=cfg.weight_decay, eps=1e-10)]


def lr_mult(step: int, total: int, cfg: Config) -> float:
    warm = max(1, int(total * cfg.warmup_frac))
    cool_start = int(total * (1 - cfg.cooldown_frac))
    if step < warm:
        return (step + 1) / warm
    if step >= cool_start:
        return max(0.0, (total - step) / max(1, total - cool_start))
    return 1.0

class Shards:
    def __init__(self, split: str, seq_len: int, seed: int):
        files = sorted(DATA_DIR.glob("train_*.bin")) if split == "train" else [DATA_DIR / "val.bin"]
        files = [f for f in files if f.exists()]
        if not files and split == "train":
            raise FileNotFoundError(f"no train shards in {DATA_DIR}; run nano/prepare_data.py")
        # the validation split is normally absent from a training process: the harness runs the exam itself
        self.data = [np.memmap(f, dtype=np.uint16, mode="r") for f in files]
        self.seq_len = seq_len
        self.rng = np.random.default_rng(seed)
        self.pos = 0

    def sample(self, n: int, device) -> tuple[torch.Tensor, torch.Tensor]:
        """Random windows (train) from a random shard."""
        d = self.data[self.rng.integers(len(self.data))]
        starts = self.rng.integers(0, len(d) - self.seq_len - 1, size=n)
        x = np.stack([d[s: s + self.seq_len] for s in starts]).astype(np.int64)
        y = np.stack([d[s + 1: s + self.seq_len + 1] for s in starts]).astype(np.int64)
        return torch.from_numpy(x).to(device, non_blocking=True), torch.from_numpy(y).to(device, non_blocking=True)

    def sequential(self, n: int, device):
        """Fixed sequential windows (val) so every run evaluates the same tokens."""
        if not self.data:
            return None
        d = self.data[0]
        T = self.seq_len
        out_x, out_y = [], []
        for _ in range(n):
            if self.pos + T + 1 > len(d):
                return None
            out_x.append(d[self.pos: self.pos + T].astype(np.int64))
            out_y.append(d[self.pos + 1: self.pos + T + 1].astype(np.int64))
            self.pos += T
        return (torch.from_numpy(np.stack(out_x)).to(device), torch.from_numpy(np.stack(out_y)).to(device))


@torch.no_grad()
def evaluate(model: GPT, cfg: Config, device, dtype) -> float:
    model.eval()
    val = Shards("val", cfg.seq_len, cfg.seed)
    n_batches = max(1, cfg.eval_tokens // (cfg.micro_batch * cfg.seq_len))
    losses = []
    for _ in range(n_batches):
        batch = val.sequential(cfg.micro_batch, device)
        if batch is None:
            break
        x, y = batch
        with torch.autocast("cuda", dtype=dtype):
            _, loss = model(x, y)
        losses.append(loss.item())
    model.train()
    return float(np.mean(losses)) if losses else float("nan")

def main() -> int:
    ap = argparse.ArgumentParser()
    for k, v in asdict(Config()).items():
        ap.add_argument(f"--{k.replace('_', '-')}", type=type(v) if not isinstance(v, bool) else int, default=v)
    ap.add_argument("--out", default=None)
    ap.add_argument("--build-only", action="store_true")
    ap.add_argument("--smoke-steps", type=int, default=0)
    ap.add_argument("--smoke-seconds", type=float, default=0.0)
    args = ap.parse_args()
    cfg = Config(**{k: getattr(args, k) for k in asdict(Config())})
    cfg.tie_embeddings = bool(cfg.tie_embeddings)

    torch.manual_seed(cfg.seed)
    np.random.seed(cfg.seed)
    torch.backends.cuda.matmul.allow_tf32 = True
    torch.backends.cudnn.allow_tf32 = True
    device = "cuda" if torch.cuda.is_available() else "cpu"
    dtype = torch.bfloat16 if cfg.dtype == "bfloat16" else torch.float16
    model = GPT(cfg).to(device)
    total, nonembed = param_counts(model)
    report = {"arch": cfg.arch, "seed": cfg.seed, "params_total": total, "params_nonembed": nonembed,
              "config": asdict(cfg)}
    print(f"[nano] params total {total/1e6:.2f}M, non-embedding {nonembed/1e6:.2f}M, device {device}", flush=True)
    out = Path(args.out) if args.out else None
    if out:
        out.mkdir(parents=True, exist_ok=True)

    def write(extra: dict) -> None:
        report.update(extra)
        if out:
            (out / "metrics.json").write_text(json.dumps(report, indent=1), encoding="utf-8")

    if args.build_only:
        write({"mode": "build"})
        print(json.dumps({k: report[k] for k in ("params_total", "params_nonembed")}))
        return 0

    opts = build_optimizers(model, cfg)
    train = Shards("train", cfg.seq_len, cfg.seed)
    accum = max(1, cfg.batch_tokens // (cfg.micro_batch * cfg.seq_len))
    total_steps = max(1, int(cfg.tokens // cfg.batch_tokens))
    if args.smoke_steps:
        total_steps = args.smoke_steps
    torch.cuda.reset_peak_memory_stats() if device == "cuda" else None
    t0 = time.time()
    losses: list[float] = []
    step = 0
    model.train()
    while step < total_steps:
        m = lr_mult(step, total_steps, cfg)
        for o in opts:
            for g in o.param_groups:
                g["lr"] = (cfg.lr if isinstance(o, Muon) else cfg.lr_adam) * m * cfg.lr_mult_both
        loss_acc = 0.0
        for _ in range(accum):
            x, y = train.sample(cfg.micro_batch, device)
            with torch.autocast("cuda", dtype=dtype):
                _, loss = model(x, y)
            (loss / accum).backward()
            loss_acc += loss.item() / accum
        if cfg.grad_clip:
            torch.nn.utils.clip_grad_norm_(model.parameters(), cfg.grad_clip)
        for o in opts:
            o.step()
        model.zero_grad(set_to_none=True)
        losses.append(loss_acc)
        step += 1
        if step % 50 == 0 or step == 1:
            el = time.time() - t0
            print(f"[nano] step {step}/{total_steps} loss {loss_acc:.4f} lr x{m:.2f} "
                  f"{step * cfg.batch_tokens / el / 1e3:.1f}k tok/s", flush=True)
        if args.smoke_seconds and time.time() - t0 > args.smoke_seconds:
            break
    wall = time.time() - t0
    peak = torch.cuda.max_memory_allocated() / 1e9 if device == "cuda" else 0.0
    res = {"steps": step, "tokens": step * cfg.batch_tokens, "wall_s": round(wall, 1),
           "peak_vram_gb": round(peak, 2), "tok_per_s": round(step * cfg.batch_tokens / max(wall, 1e-6)),
           "loss_first": round(float(np.mean(losses[:5])), 4) if losses else None,
           "loss_last": round(float(np.mean(losses[-5:])), 4) if losses else None,
           "loss_curve": [round(float(l), 4) for l in losses]}   # per-step curve: matched-step comparisons
    if args.smoke_steps or args.smoke_seconds:
        res["mode"] = "smoke"
        write(res)
        print(json.dumps(res))
        return 0
    if out:                                   # the harness's exam (nano/exam.py) reads these weights
        torch.save({"state_dict": model.state_dict(), "config": asdict(cfg)}, out / "model.pt")
    val_loss = evaluate(model, cfg, device, dtype)   # a self-report; absent when the split is hidden
    has_val = val_loss == val_loss
    res.update({"mode": "full", "val_loss": round(val_loss, 4) if has_val else None,
                "val_bpb": round(val_loss / math.log(2) * 0.2446, 4) if has_val else None})   # tokens/byte ~0.2446 for GPT-2 BPE on English web text (approx)
    write(res)
    print(json.dumps(res))
    return 0


if __name__ == "__main__":
    raise SystemExit(main())

THE IDEA:
IDEA: Direction-only tie with per-token scale and bias
MECHANISM: Keep the shared 50304x384 table but use its RMS-normalized rows as the output classifier, multiplied by a learned per-token log-scale and plus a learned per-token bias (two vectors of 50304 scalars, zero-init). The tie currently forces one row norm to serve as input salience and as output frequency prior at once, and the 2026 tying analysis shows the output gradient wins that fight; decoupling norm from direction lets the shared direction serve both roles while rare rows (rank 10k+, 11.6 nats) keep a usable direction even when decay shrinks their norm. The conflict is intrinsic to tying and do
PREDICTION: Loss curve pulls below baseline within the first few hundred steps as the bias absorbs the unigram prior, and the gap persists; parameters rise by about 100k to roughly 21.34M; step time within 2 percent (one 50304x384 normalize per forward, done in bf16 to stay under the 0.1 GB memory headroom); th
TEST: Build-only for the count, 3 steps for time and peak memory, then 45 matched steps against baseline and against the bias-only variant below to isolate the normalization.
TARGET: loss on rare tokens, down
KIND: leap (an ambitious idea: build it faithfully, do not water it down into a known technique)
BREAKS: The tied row's norm must encode both input salience and output frequency; this flips it so norm is output-only and lives in two cheap scalar vectors.
KEEPS: One shared table, plain next-token cross entropy, final RMSNorm, mean read-out, the same Muon and AdamW grouping (the two new vectors fall to AdamW as 1-D parameters).
KILL: Mean training loss over matched steps 30 to 45 not at least 0.02 nats below baseline, or below the bias-only variant by less than 0.005.
VARIANTS OF THE SAME IDEA FROM OTHER CONTEXTS (choose the version you judge most likely to work):
- Direction-only tied head with a counted unigram bias: The brief's centering paper identifies output anisotropy, and the rare-token loss equals the unigram cost; literalise "row norm equals log frequency" and the consequence is that the tied table spends its norm, which the input lookup also sees, on a quantity a bias can hold. Logits become g·norm(x)·(E/‖E‖_row)ᵀ + b, with b initialised to log counts from one train shard (stored as a parameter so the exam sees it) and g a learned scalar starting at the baseline row norm.
- Unigram bias on the tied head: Add a learned 50304-vector bias to the logits, zero-init, carried by the fused linear call. Without it the tied table has to encode token frequency inside the embedding geometry, which pollutes the input side that the same rows feed; a bias removes that burden for free and should give most of its value early, when the model is still learning the unigram and bigram statistics that dominate the first 50M tokens.
- Frozen log-unigram prior in the logits: Written as a code length, the loss is a two-part code, prior plus conditional, and the net currently pays for the prior by growing frequent rows and shrinking rare ones under weight decay, so rare tokens at rank 10k+ sit at 11.6 nats with 12 percent of the tokens. The natural edit is to add a fixed per-token log count, computed once from a 50M-token sample of the train shards at startup with add-one smoothing, as a persistent buffer added to the logits. Rows then carry only conditional information, rare rows stop being pushed to zero norm, and the first 100 steps are not spent learning the uni

- `--smoke-steps 45` gives warm = 1 and cool_start = 31; the full run gives warm = 30 and cool_start = 1067 out of 1525 steps. Steps 30–45 have different LR in the two modes.

- `Shards` uses its own numpy rng seeded by `--seed`, independent of torch, so same seed means the same batches in every arm.
- Losses go through `log_softmax`, which is on autocast's float32 list, so the logged loss is fp32 (https://pytorch.org/docs/stable/amp.html).
- PyTorch documents several CUDA backward ops as nondeterministic, so same-seed reruns may diverge (https://pytorch.org/docs/stable/notes/randomness.html).
- `clip_grad_norm_` (clip at 1.0) covers all parameters, including any new bias.
- Scale-invariant weights trained with decay grow their norm through tangential steps until decay balances it (https://arxiv.org/abs/2006.08419). AdamW's per-coordinate normalisation also leaves a radial component, so L3's "norm set only by input and decay" is incomplete.
- Three baseline seeds already exist with loss curves (from the layer-0 gradient measurement). Computing their across-seed sd of `mean(loss_curve[29:45])` is free.
- The head is `nn.Linear` with weight tied to `tok.weight`, applied to `self.norm(x.mean(dim=2))`. Logits stay bf16 under autocast and the loss is plain F.cross_entropy.
- `param_counts` subtracts only `tok.weight`, so s and b (2 × 50304 = 100,608) count as non-embedding: 21.24M → about 21.34M, cap about 23.36M. A frozen buffer would not count.
- `named_parameters` dedupes the tie, so only `tok.weight` appears; it goes to AdamW (lr 3e-3, wd 0.1, betas 0.9/0.95). New 1-D vectors fall to AdamW too, but need their own wd = 0 group.
- Steps = 50e6 / 32768 = 1525; accum = 8 micro-batches of 4×1024; warmup 2%, cooldown 30%.
- Initial row RMS is 0.02 and RMSNorm output RMS is 1. Initialise s = log(0.02) for RMS-normalised rows so step-0 logits match baseline.
- Logits tensor per micro-batch is 4096×50304 bf16 ≈ 412 MB. Never add a logits-sized elementwise op; fold s into the weight and b into F.linear.
- The exam builds GPT(cfg) and loads `state_dict`. The prior must be a Parameter or persistent buffer filled in main before training, never computed in `__init__`.
- Mean over Sinkhorn-mixed streams preserves the input, so the readout always contains exactly e_x plus branch writes.
