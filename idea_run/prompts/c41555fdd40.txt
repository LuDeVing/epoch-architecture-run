class RMSNorm(nn.Module):
    def __init__(self, d: int, eps: float = 1e-6):
        super().__init__()
        self.weight = nn.Parameter(torch.ones(d))
        self.eps = eps

    def forward(self, x):
        return F.rms_norm(x, (x.size(-1),), self.weight, self.eps)


def rope_cache(seq_len: int, head_dim: int, theta: float, device) -> tuple[torch.Tensor, torch.Tensor]:
    inv = 1.0 / (theta ** (torch.arange(0, head_dim, 2, device=device).float() / head_dim))
    t = torch.arange(seq_len, device=device).float()
    freqs = torch.outer(t, inv)                       # (T, D/2)
    return freqs.cos(), freqs.sin()


def apply_rope(x: torch.Tensor, cos: torch.Tensor, sin: torch.Tensor) -> torch.Tensor:
    # x: (B, H, T, D); rotate pairs (x1, x2) = (x[..., :D/2], x[..., D/2:])
    d = x.size(-1) // 2
    x1, x2 = x[..., :d], x[..., d:]
    cos, sin = cos[: x.size(2)].to(x.dtype), sin[: x.size(2)].to(x.dtype)
    return torch.cat([x1 * cos - x2 * sin, x1 * sin + x2 * cos], dim=-1)


class Attention(nn.Module):
    def __init__(self, cfg: Config):
        super().__init__()
        self.n_head = cfg.n_head
        self.hd = cfg.d_model // cfg.n_head
        self.rd = 32
        self.wq = nn.Linear(cfg.d_model, self.n_head * (self.hd + self.rd), bias=False)
        self.compress = nn.Linear(cfg.d_model, 128, bias=False)
        self.latent_norm = RMSNorm(128)
        self.expand = nn.Linear(128, 2 * self.n_head * self.hd, bias=False)
        self.rotary = nn.Linear(cfg.d_model, self.rd, bias=False)
        self.q_norm, self.k_norm = RMSNorm(self.hd), RMSNorm(self.hd)
        self.wo = nn.Linear(self.n_head * self.hd, cfg.d_model, bias=False)
        self.window = cfg.sliding_window

    def forward(self, x, cos, sin):
        B, T, _ = x.shape
        q = self.wq(x).view(B, T, self.n_head, self.hd + self.rd).transpose(1, 2)
        qc, qr = q.split([self.hd, self.rd], dim=-1)
        kv = self.expand(self.latent_norm(self.compress(x)))
        k, v = kv.view(B, T, self.n_head, 2 * self.hd).transpose(1, 2).chunk(2, dim=-1)
        kr = self.rotary(x).view(B, T, 1, self.rd).transpose(1, 2)
        qr, kr = apply_rope(qr, cos, sin), apply_rope(kr, cos, sin)
        q = torch.cat([self.q_norm(qc), qr], dim=-1)
        k = torch.cat([self.k_norm(k), kr.expand(-1, self.n_head, -1, -1)], dim=-1)
        if self.window and self.window < T:
            i = torch.arange(T, device=x.device)
            mask = (i[None, :] <= i[:, None]) & (i[:, None] - i[None, :] < self.window)
            y = F.scaled_dot_product_attention(q, k, v, attn_mask=mask)
        else:
            y = F.scaled_dot_product_attention(q, k, v, is_causal=True)
        return self.wo(y.transpose(1, 2).reshape(B, T, -1))


class MLP(nn.Module):
    """SwiGLU feed-forward."""
    def __init__(self, cfg: Config):
        super().__init__()
        self.w_gate = nn.Linear(cfg.d_model, cfg.d_ff, bias=False)
        self.w_up = nn.Linear(cfg.d_model, cfg.d_ff, bias=False)
        self.w_down = nn.Linear(cfg.d_ff, cfg.d_model, bias=False)

    def forward(self, x):
        return self.w_down(F.silu(self.w_gate(x)) * self.w_up(x))


class Block(nn.Module):
    def __init__(self, cfg: Config):
        super().__init__()
        self.n1, self.n2 = RMSNorm(cfg.d_model), RMSNorm(cfg.d_model)
        self.attn, self.mlp = Attention(cfg), MLP(cfg)
        self.mix_logits = nn.Parameter(torch.eye(4).repeat(2, 1, 1) * 6.0)
        self.read_logits = nn.Parameter(torch.zeros(2, 4))
        self.write_logits = nn.Parameter(torch.tensor([[0.03, -0.01, -0.01, -0.01], [-0.01, 0.03, -0.01, -0.01]]))

    def combine(self, x, branch, stage):
        mixing = self.mix_logits[stage].float().exp()
        for _ in range(20):
            mixing = mixing / mixing.sum(dim=-1, keepdim=True)
            mixing = mixing / mixing.sum(dim=-2, keepdim=True)
        mixed = torch.einsum('ij,btjd->btid', mixing.to(x.dtype), x)
        weights = 2 * self.write_logits[stage].sigmoid()
        return mixed + branch.unsqueeze(2) * weights.to(x.dtype)[None, None, :, None]

    def forward(self, x, cos, sin):
        weights = self.read_logits[0].softmax(dim=-1).to(x.dtype)
        branch = self.attn(self.n1((x * weights[None, None, :, None]).sum(2)), cos, sin)
        x = self.combine(x, branch, 0)
        weights = self.read_logits[1].softmax(dim=-1).to(x.dtype)
        branch = self.mlp(self.n2((x * weights[None, None, :, None]).sum(2)))
        return self.combine(x, branch, 1)


class GPT(nn.Module):
    def __init__(self, cfg: Config):
        super().__init__()
        self.cfg = cfg
        self.tok = nn.Embedding(cfg.vocab_size, cfg.d_model)
        self.blocks = nn.ModuleList([Block(cfg) for _ in range(cfg.n_layer)])
        self.norm = RMSNorm(cfg.d_model)
        self.head = nn.Linear(cfg.d_model, cfg.vocab_size, bias=False)
        if cfg.tie_embeddings:
            self.head.weight = self.tok.weight
        self.apply(self._init)
        for b in self.blocks:                              # scaled residual-branch init
            nn.init.normal_(b.attn.wo.weight, std=0.02 / math.sqrt(2 * cfg.n_layer))
            nn.init.normal_(b.mlp.w_down.weight, std=0.02 / math.sqrt(2 * cfg.n_layer))

    @staticmethod
    def _init(m):
        if isinstance(m, nn.Linear):
            nn.init.normal_(m.weight, std=0.02)
        elif isinstance(m, nn.Embedding):
            nn.init.normal_(m.weight, std=0.02)

    def forward(self, idx, targets=None):
        B, T = idx.shape
        cos, sin = rope_cache(T, 32, self.cfg.rope_theta, idx.device)
        x = self.tok(idx).unsqueeze(2).expand(-1, -1, 4, -1)
        for b in self.blocks:
            x = b(x, cos, sin)
        logits = self.head(self.norm(x.mean(dim=2)))
        if targets is None:
            return logits, None
        # logits stay in the autocast dtype; a float32 copy of (B*T, 50304) alone is 0.8 GB at B=4
        loss = F.cross_entropy(logits.view(-1, logits.size(-1)), targets.view(-1))
        return logits, loss


def param_counts(model: nn.Module) -> tuple[int, int]:
    total = sum(p.numel() for p in model.parameters())
    embed = model.tok.weight.numel()
    return total, total - embed


# ======================================================================================
# Muon optimizer (Newton-Schulz orthogonalized momentum; Jordan et al. 2024)
# ======================================================================================
def newton_schulz5(G: torch.Tensor, steps: int = 5) -> torch.Tensor:
    a, b, c = (3.4445, -4.7750, 2.0315)
    X = G.bfloat16()
    X = X / (X.norm() + 1e-7)
    transposed = G.size(0) > G.size(1)
    if transposed:
        X = X.T
    for _ in range(steps):
        A = X @ X.T
        B_ = b * A + c * A @ A
        X = a * X + B_ @ X
    return (X.T if transposed else X).to(G.dtype)


class Muon(torch.optim.Optimizer):
    def __init__(self, params, lr=0.02, momentum=0.95, weight_decay=0.0, nesterov=True):
        super().__init__(params, dict(lr=lr, momentum=momentum, weight_decay=weight_decay, nesterov=nesterov))

    @torch.no_grad()
    def step(self):
        for group in self.param_groups:
            for p in group["params"]:
                if p.grad is None:
                    continue
                g = p.grad
                st = self.state[p]
                if "buf" not in st:
                    st["buf"] = torch.zeros_like(g)
                buf = st["buf"]
                buf.mul_(group["momentum"]).add_(g)
                g = g.add(buf, alpha=group["momentum"]) if group["nesterov"] else buf
                u = newton_schulz5(g.reshape(g.size(0), -1)).view_as(g)
                scale = max(1.0, g.size(0) / g.size(1)) ** 0.5     # shape-aware scaling
                if group["weight_decay"]:
                    p.mul_(1 - group["lr"] * group["weight_decay"])
                p.add_(u, alpha=-group["lr"] * scale)


def build_optimizers(model: GPT, cfg: Config):
    matrix = [p for n, p in model.named_parameters() if p.ndim == 2 and "tok" not in n and "head" not in n]
    other = [p for n, p in model.named_parameters() if not (p.ndim == 2 and "tok" not in n and "head" not in n)]
    return [Muon(matrix, lr=cfg.lr, weight_decay=cfg.weight_decay),
            torch.optim.AdamW(other, lr=cfg.lr_adam, betas=(0.9, 0.95), weight_decay=cfg.weight_decay, eps=1e-10)]


def lr_mult(step: int, total: int, cfg: Config) -> float:
    warm = max(1, int(total * cfg.warmup_frac))
    cool_start = int(total * (1 - cfg.cooldown_frac))
    if step < warm:
        return (step + 1) / warm
    if step >= cool_start:
        return max(0.0, (total - step) / max(1, total - cool_start))
    return 1.0

class Shards:
    def __init__(self, split: str, seq_len: int, seed: int):
        files = sorted(DATA_DIR.glob("train_*.bin")) if split == "train" else [DATA_DIR / "val.bin"]
        files = [f for f in files if f.exists()]
        if not files and split == "train":
            raise FileNotFoundError(f"no train shards in {DATA_DIR}; run nano/prepare_data.py")
        # the validation split is normally absent from a training process: the harness runs the exam itself
        self.data = [np.memmap(f, dtype=np.uint16, mode="r") for f in files]
        self.seq_len = seq_len
        self.rng = np.random.default_rng(seed)
        self.pos = 0

    def sample(self, n: int, device) -> tuple[torch.Tensor, torch.Tensor]:
        """Random windows (train) from a random shard."""
        d = self.data[self.rng.integers(len(self.data))]
        starts = self.rng.integers(0, len(d) - self.seq_len - 1, size=n)
        x = np.stack([d[s: s + self.seq_len] for s in starts]).astype(np.int64)
        y = np.stack([d[s + 1: s + self.seq_len + 1] for s in starts]).astype(np.int64)
        return torch.from_numpy(x).to(device, non_blocking=True), torch.from_numpy(y).to(device, non_blocking=True)

    def sequential(self, n: int, device):
        """Fixed sequential windows (val) so every run evaluates the same tokens."""
        if not self.data:
            return None
        d = self.data[0]
        T = self.seq_len
        out_x, out_y = [], []
        for _ in range(n):
            if self.pos + T + 1 > len(d):
                return None
            out_x.append(d[self.pos: self.pos + T].astype(np.int64))
            out_y.append(d[self.pos + 1: self.pos + T + 1].astype(np.int64))
            self.pos += T
        return (torch.from_numpy(np.stack(out_x)).to(device), torch.from_numpy(np.stack(out_y)).to(device))


@torch.no_grad()
def evaluate(model: GPT, cfg: Config, device, dtype) -> float:
    model.eval()
    val = Shards("val", cfg.seq_len, cfg.seed)
    n_batches = max(1, cfg.eval_tokens // (cfg.micro_batch * cfg.seq_len))
    losses = []
    for _ in range(n_batches):
        batch = val.sequential(cfg.micro_batch, device)
        if batch is None:
            break
        x, y = batch
        with torch.autocast("cuda", dtype=dtype):
            _, loss = model(x, y)
        losses.append(loss.item())
    model.train()
    return float(np.mean(losses)) if losses else float("nan")

def main() -> int:
    ap = argparse.ArgumentParser()
    for k, v in asdict(Config()).items():
        ap.add_argument(f"--{k.replace('_', '-')}", type=type(v) if not isinstance(v, bool) else int, default=v)
    ap.add_argument("--out", default=None)
    ap.add_argument("--build-only", action="store_true")
    ap.add_argument("--smoke-steps", type=int, default=0)
    ap.add_argument("--smoke-seconds", type=float, default=0.0)
    args = ap.parse_args()
    cfg = Config(**{k: getattr(args, k) for k in asdict(Config())})
    cfg.tie_embeddings = bool(cfg.tie_embeddings)

    torch.manual_seed(cfg.seed)
    np.random.seed(cfg.seed)
    torch.backends.cuda.matmul.allow_tf32 = True
    torch.backends.cudnn.allow_tf32 = True
    device = "cuda" if torch.cuda.is_available() else "cpu"
    dtype = torch.bfloat16 if cfg.dtype == "bfloat16" else torch.float16
    model = GPT(cfg).to(device)
    total, nonembed = param_counts(model)
    report = {"arch": cfg.arch, "seed": cfg.seed, "params_total": total, "params_nonembed": nonembed,
              "config": asdict(cfg)}
    print(f"[nano] params total {total/1e6:.2f}M, non-embedding {nonembed/1e6:.2f}M, device {device}", flush=True)
    out = Path(args.out) if args.out else None
    if out:
        out.mkdir(parents=True, exist_ok=True)

    def write(extra: dict) -> None:
        report.update(extra)
        if out:
            (out / "metrics.json").write_text(json.dumps(report, indent=1), encoding="utf-8")

    if args.build_only:
        write({"mode": "build"})
        print(json.dumps({k: report[k] for k in ("params_total", "params_nonembed")}))
        return 0

    opts = build_optimizers(model, cfg)
    train = Shards("train", cfg.seq_len, cfg.seed)
    accum = max(1, cfg.batch_tokens // (cfg.micro_batch * cfg.seq_len))
    total_steps = max(1, int(cfg.tokens // cfg.batch_tokens))
    if args.smoke_steps:
        total_steps = args.smoke_steps
    torch.cuda.reset_peak_memory_stats() if device == "cuda" else None
    t0 = time.time()
    losses: list[float] = []
    step = 0
    model.train()
    while step < total_steps:
        m = lr_mult(step, total_steps, cfg)
        for o in opts:
            for g in o.param_groups:
                g["lr"] = (cfg.lr if isinstance(o, Muon) else cfg.lr_adam) * m * cfg.lr_mult_both
        loss_acc = 0.0
        for _ in range(accum):
            x, y = train.sample(cfg.micro_batch, device)
            with torch.autocast("cuda", dtype=dtype):
                _, loss = model(x, y)
            (loss / accum).backward()
            loss_acc += loss.item() / accum
        if cfg.grad_clip:
            torch.nn.utils.clip_grad_norm_(model.parameters(), cfg.grad_clip)
        for o in opts:
            o.step()
        model.zero_grad(set_to_none=True)
        losses.append(loss_acc)
        step += 1
        if step % 50 == 0 or step == 1:
            el = time.time() - t0
            print(f"[nano] step {step}/{total_steps} loss {loss_acc:.4f} lr x{m:.2f} "
                  f"{step * cfg.batch_tokens / el / 1e3:.1f}k tok/s", flush=True)
        if args.smoke_seconds and time.time() - t0 > args.smoke_seconds:
            break
    wall = time.time() - t0
    peak = torch.cuda.max_memory_allocated() / 1e9 if device == "cuda" else 0.0
    res = {"steps": step, "tokens": step * cfg.batch_tokens, "wall_s": round(wall, 1),
           "peak_vram_gb": round(peak, 2), "tok_per_s": round(step * cfg.batch_tokens / max(wall, 1e-6)),
           "loss_first": round(float(np.mean(losses[:5])), 4) if losses else None,
           "loss_last": round(float(np.mean(losses[-5:])), 4) if losses else None,
           "loss_curve": [round(float(l), 4) for l in losses]}   # per-step curve: matched-step comparisons
    if args.smoke_steps or args.smoke_seconds:
        res["mode"] = "smoke"
        write(res)
        print(json.dumps(res))
        return 0
    if out:                                   # the harness's exam (nano/exam.py) reads these weights
        torch.save({"state_dict": model.state_dict(), "config": asdict(cfg)}, out / "model.pt")
    val_loss = evaluate(model, cfg, device, dtype)   # a self-report; absent when the split is hidden
    has_val = val_loss == val_loss
    res.update({"mode": "full", "val_loss": round(val_loss, 4) if has_val else None,

## The strongest open models now
- Beam, Reflection AI, announced 2026-10-05, weights and report promised "later this month": 501B total, 23B active; interleaved local/global attention, fine-grained routed experts, a "controlled residual stream" (no further detail) (https://techcrunch.com/2026/10/05/reflection-debuts-beam-a-open-weight-ai-model-to-rival-chinese-models-at-lower-compute-cost/)
- Kimi K3, Moonshot, Jul 2026: 2.8T total, 104B active; 93 layers, 69 Kimi Delta Attention (linear) + 24 gated MLA, 896 experts (16 routed + 2 shared) in a 3584-d latent space, Attention Residuals, 1M context (https://github.com/MoonshotAI/Kimi-K3)
- Inkling, Thinking Machines, Jul 2026: 975B total, 41B active; 66 layers at 5:1 sliding(512)/global, no RoPE (learned input-dependent relative-position bias), four causal short convs per layer (https://www.sebastianraschka.com/blog/2026/inkling-architecture-benchmark-notes.html)
- Gemma 4, Google, Jul 2026: dense 12B/31B and 26B-A4B (3.8B active); 5:1 local/global, p-RoPE on global layers, cross-layer KV sharing in E2B/E4B, MTP drafter (https://arxiv.org/html/2607.02770)
- Tencent Hy4-preview, 770B total, 49B active: gated DeepSeek-style sparse attention with cross-layer index reuse, 4-stream identity hyper-connections, 256 routed + 1 shared expert (https://mer.vin/news/tencents-hy4-preview-edges-out-glm-5-3-and-kimi-k3/)
- Nemotron 3 Ultra, NVIDIA, Jun 2026: 550B total, 55B active; 108 layers = 48 Mamba-2 + 48 LatentMoE + 12 GQA attention (https://sebastianraschka.com/blog/2026/nemotron-3-ultra-latent-moe.html)
- MiniMax M3, 2026-06-01: about 428B total, 23B active (one source says 22B); block-sparse attention (MSA) on a GQA backbone with full, uncompressed KV (https://www.marktechpost.com/2026/06/01/minimax-releases-minimax-m3-with-msa-architecture-supporting-1m-token-context-native-multimodality-and-agentic-coding/)
- DeepSeek-V4-Pro/Flash, Jun 2026: 1.6T/49B and 284B/13B; interleaved compressed-sparse and heavily-compressed attention, mHC residuals, hash-routed early MoE layers (https://arxiv.org/html/2606.19348v1)
- GLM-5.2/5.3-Flash, Z.ai: 744B/40B with DSA plus IndexShare; the Flash model (320B/18B) is the first GLM linear-attention hybrid, with mHC (https://github.com/zai-org/GLM-5)
- Qwen3.5-397B-A17B (Qwen3.6-35B-A3B and 27B dense followed in Apr 2026): 60 layers, 45 Gated DeltaNet + 15 gated softmax-attention (https://sebastianraschka.com/llm-architecture-gallery/gated-attention/)
- MiMo-V2-Flash, Xiaomi, Jan 2026: 309B/15B; 5:1 sliding(128)/global, learnable sink bias, 256 experts with 8 active and no shared expert, 3-layer MTP (https://arxiv.org/html/2601.02780v1)
- Olmo Hybrid, Ai2, Apr 2026: 7B; Olmo 3 7B with its sliding-window layers replaced by Gated DeltaNet (https://arxiv.org/abs/2604.03444)

## Components, with measured effects
- Linear/full hybrid: Gated DeltaNet 3:1 (Qwen3-Next/3.5, Olmo Hybrid, Solar Open 2 uses Kimi Delta Attention 3:1), KDA+MLA (Kimi Linear, K3), Mamba-2 (Nemotron). Kimi Linear, same recipe: beats full MLA, up to 75% less KV, up to 6x decode at 1M. Olmo Hybrid reaches Olmo 3's MMLU with 49% fewer tokens (https://alphaxiv.org/abs/2510.26692; https://allenai.org/blog/olmohybrid)
- Learned top-k sparse attention (DSA): GLM-5, Hy4. GLM-5.2 IndexShare reuses one indexer across 4 sparse layers: per-token FLOPs cut 2.9x at 1M (https://github.com/zai-org/GLM-5)
- Compressed KV attention (V4): CSA compresses every m tokens (m=4 per Raschka) with top-k selection and shared-KV MQA; HCA compresses 128 tokens per entry, dense. Both add a 128-token window branch. V4-Pro uses 27% of V3.2's FLOPs and 10% of its KV at 1M; Flash 10% and 7% (https://arxiv.org/html/2606.19348v1)
- Block-sparse attention (MSA): an index branch scores KV blocks, then exact attention runs on top-k blocks per GQA group. On a 109B testbed: 28.4x less attention compute at 1M, "on par with GQA", 14.2x prefill and 7.6x decode on H800 (https://arxiv.org/abs/2606.13392)
- Sliding-window/global interleave: gpt-oss 1:1 (W=128), Step 3.5 3:1, Gemma 4 and Inkling and MiMo 5:1. MiMo ablation (32B dense): W=128 beats W=512 on RULER-32k (89.4 vs 84.7) and MRCR (34.4 vs 19.6) at equal MMLU 58.3 (https://arxiv.org/html/2601.02780v1)
- Attention sink (learned logit in softmax denominator): gpt-oss, MiMo, V4. MiMo ablation: MMLU 54.9 without vs 58.3 with sink (https://arxiv.org/html/2601.02780v1)
- Attention output gate: Qwen3-Next/3.5, Hy4, Laguna. A 30-variant study found a post-SDPA sigmoid gate best: fewer sinks, better long-context extrapolation, under 2% latency (https://sebastianraschka.com/llm-architecture-gallery/gated-attention/)
- Cross-layer KV sharing and key=value on global layers (Gemma 4): last 20 of 35 (E2B) and 18 of 42 (E4B) layers reuse KV; global KV cache down up to 37.5%. No quality ablation reported (https://arxiv.org/html/2607.02770)
- Positional scheme: partial RoPE (V4 last 64 dims, MiMo first 64), p-RoPE p=0.25 on Gemma 4 global layers, NoPE in Kimi Linear's MLA layers, none in Inkling (relative-position bias). Only MiMo's cost-free sink/window ablations above are measured here; no RoPE ablation found (https://arxiv.org/html/2607.02770; https://magazine.sebastianraschka.com/p/the-big-llm-architecture-comparison)
- QK normalisation: V4 RMSNorms query heads and the compressed KV head and says this removes the need for QK-Clip. Gemma 4 also uses QKNorm (https://arxiv.org/html/2606.19348v1)
- Short causal convs: Inkling, kernel 4 after K and V projections and on both branch outputs before the residual add. The post notes no ablation isolates them (https://www.sebastianraschka.com/blog/2026/inkling-architecture-benchmark-notes.html)
- Fine-grained MoE (routed/total + shared): Hy4 8/256+1, Step 3.5 8/288+1, Gemma 4 8/128+1, MiMo 8/256+0, Inkling 6/256+2, K3 16/896+2, gpt-oss 4/128, Nemotron 22/512+shared (https://hf.co/stepfun-ai/Step-3.5-Flash-Base-Midtrain; https://arxiv.org/html/2601.02780v1)
- MoE routing (V4): sqrt(softplus) affinity replaces sigmoid, aux-loss-free balancing plus a slight sequence-wise loss, node-limit constraint removed, hash-by-token-ID routing in the first MoE blocks. No ablation in the text read (https://arxiv.org/html/2606.19348v1)
- LatentMoE: routing and experts run in a down-projected space (Nemotron 8192→2048, 4x; K3 latent 3584). Report says routed parameter loads and all-to-all fall by d/ℓ (about 4x), but layer speedup is below 4x (https://arxiv.org/html/2512.20856v1; https://sebastianraschka.com/blog/2026/nemotron-3-ultra-latent-moe.html)
- Multi-stream residual (mHC): V4 expands the residual to n=4 streams with a doubly stochastic mixing matrix (Sinkhorn, 20 iterations). Overhead is 6.7% of the overlapped pipeline stage; no quality ablation in the text read. Hy4 uses identity hyper-connections, GLM-5.3-Flash uses mHC (https://arxiv.org/html/2606.19348v1)
- Attention Residuals (K3): each layer takes a softmax-attention-weighted mix of earlier layer outputs, with a block variant. On Kimi Linear 48B/3B at 1.4T tokens: more uniform output magnitude and gradients across depth, better on all evaluated tasks (https://arxiv.org/pdf/2603.15031.pdf)
- Norm placement: Gemma 4 pre+post RMSNorm; Olmo 3 post-norm; Inkling adds an RMSNorm directly after the embedding lookup. No measured effects found (https://sebastianraschka.com/blog/2026/gemma-4-release-notes.html)
- Multi-token prediction: V4 reuses V3's MTP; MiMo's block is dense FFN + sliding-window attention, 0.33B each. With 3 layers, acceptance length reaches about 3.6 and the speedup is 2.39x (table range 1.86-2.70x). GLM-5.2's improved MTP gains up to 20% acceptance. Nemotron has 2 shared-parameter heads. Gemma 4 has a separate drafter (https://arxiv.org/html/2601.02780v1)
- Embeddings and head: Gemma 4 E2B/E4B add per-layer embeddings (vocab 262k); the drafter's head drops from d×262k to d×4096 via token clusters with similar acceptance. K3 vocab is 160K (https://arxiv.org/html/2607.02770)
- Depth reuse: Nanbeige 4.2 3B runs its 22-layer stack twice; two passes kept about 75% of standard token efficiency and more passes gave little gain (https://sebastianraschka.com/blog/2026/notable-open-weight-models-this-week.html)
- Per-layer head budgeting (Laguna XS.2): 8 KV heads fixed; 6 query heads per KV head on global layers, 8 on sliding layers (https://magazine.sebastianraschka.com/p/recent-developments-in-llm-architectures)

## Not settled
- Linear (Qwen, Kimi, Olmo, Nemotron, Solar), learned-sparse (DeepSeek, GLM, Hy4, MiniMax M3) and windowed (Gemma, MiMo, gpt-oss) attention are all in use. MiniMax M2 used full attention, and MiniMax says MSA matched full attention on "the vast majority of capabilities" without per-task numbers (https://www.minimax.io/blog/minimax-m3)
- Many headline components have no isolated ablation in their reports: Gemma 4 reports none, and Inkling's convs, bias and embedding norm are unablated. The V4 text I read has no mHC quality ablation, and K3's "2.5x scaling efficiency over K2" comes with no methodology (https://arxiv.org/html/2607.02770)
- Residual depth mixing has three competing designs (mHC, identity hyper-connections, Attention Residuals), with no cross-comparison found (https://arxiv.org/pdf/2603.15031.pdf)
- Shared experts range from none (MiMo) to 1-2 (Gemma, Step, K3, Inkling), and positional schemes range from partial RoPE to p-RoPE to NoPE to learned bias. Sources disagree on some counts (MiniMax M3 active 22B vs 23B; gpt-oss router softmax vs sigmoid) (https://arxiv.org/html/2601.02780v1)
- Efficiency figures mix architecture with serving stacks, e.g. Nemotron 3 Ultra's 5.9x/4.8x/1.6x throughput over GLM-5.1/Kimi-K2.6/Qwen3.5, run on TensorRT-LLM against vLLM (https://sebastianraschka.com/blog/2026/nemotron-3-ultra-latent-moe.html)
